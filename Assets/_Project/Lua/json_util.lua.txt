local M = {}

local function skip_ws(str, pos)
    while pos <= #str do
        local c = str:sub(pos, pos)
        if c == " " or c == "\t" or c == "\n" or c == "\r" then
            pos = pos + 1
        else
            break
        end
    end
    return pos
end

local function utf8_char(code)
    if code < 0x80 then
        return string.char(code)
    elseif code < 0x800 then
        return string.char(0xC0 + math.floor(code / 0x40), 0x80 + code % 0x40)
    elseif code < 0x10000 then
        return string.char(
            0xE0 + math.floor(code / 0x1000),
            0x80 + math.floor(code / 0x40) % 0x40,
            0x80 + code % 0x40)
    else
        return string.char(
            0xF0 + math.floor(code / 0x40000),
            0x80 + math.floor(code / 0x1000) % 0x40,
            0x80 + math.floor(code / 0x40) % 0x40,
            0x80 + code % 0x40)
    end
end

local function parse_string(str, pos)
    pos = pos + 1
    local out = {}
    while pos <= #str do
        local c = str:sub(pos, pos)
        if c == '"' then
            return table.concat(out), pos + 1
        elseif c == "\\" then
            local esc = str:sub(pos + 1, pos + 1)
            if esc == '"' or esc == "\\" or esc == "/" then
                out[#out + 1] = esc
                pos = pos + 2
            elseif esc == "b" then
                out[#out + 1] = "\b"
                pos = pos + 2
            elseif esc == "f" then
                out[#out + 1] = "\f"
                pos = pos + 2
            elseif esc == "n" then
                out[#out + 1] = "\n"
                pos = pos + 2
            elseif esc == "r" then
                out[#out + 1] = "\r"
                pos = pos + 2
            elseif esc == "t" then
                out[#out + 1] = "\t"
                pos = pos + 2
            elseif esc == "u" then
                local code = tonumber(str:sub(pos + 2, pos + 5), 16)
                if code then
                    out[#out + 1] = utf8_char(code)
                    pos = pos + 6
                else
                    out[#out + 1] = esc
                    pos = pos + 2
                end
            else
                out[#out + 1] = esc
                pos = pos + 2
            end
        else
            out[#out + 1] = c
            pos = pos + 1
        end
    end
    error("unterminated string")
end

local function parse_number(str, pos)
    local start = pos
    if str:sub(pos, pos) == "-" then
        pos = pos + 1
    end
    while pos <= #str do
        local c = str:sub(pos, pos)
        if (c >= "0" and c <= "9") or c == "." or c == "e" or c == "E" or c == "+" or c == "-" then
            pos = pos + 1
        else
            break
        end
    end
    return tonumber(str:sub(start, pos - 1)), pos
end

local function parse_value(str, pos)
    pos = skip_ws(str, pos)
    local c = str:sub(pos, pos)

    if c == "{" then
        local obj = {}
        pos = skip_ws(str, pos + 1)
        if str:sub(pos, pos) == "}" then
            return obj, pos + 1
        end
        while true do
            pos = skip_ws(str, pos)
            local key, next_pos = parse_string(str, pos)
            pos = skip_ws(str, next_pos)
            if str:sub(pos, pos) ~= ":" then
                error("expected colon")
            end
            local value, value_pos = parse_value(str, skip_ws(str, pos + 1))
            obj[key] = value
            pos = skip_ws(str, value_pos)
            local sep = str:sub(pos, pos)
            if sep == "," then
                pos = pos + 1
            elseif sep == "}" then
                return obj, pos + 1
            else
                error("expected comma or }")
            end
        end
    elseif c == "[" then
        local arr = {}
        pos = skip_ws(str, pos + 1)
        if str:sub(pos, pos) == "]" then
            return arr, pos + 1
        end
        while true do
            local value, value_pos = parse_value(str, pos)
            arr[#arr + 1] = value
            pos = skip_ws(str, value_pos)
            local sep = str:sub(pos, pos)
            if sep == "," then
                pos = pos + 1
            elseif sep == "]" then
                return arr, pos + 1
            else
                error("expected comma or ]")
            end
        end
    elseif c == '"' then
        return parse_string(str, pos)
    elseif c == "t" then
        if str:sub(pos, pos + 3) == "true" then
            return true, pos + 4
        end
        error("invalid token")
    elseif c == "f" then
        if str:sub(pos, pos + 4) == "false" then
            return false, pos + 5
        end
        error("invalid token")
    elseif c == "n" then
        if str:sub(pos, pos + 3) == "null" then
            return nil, pos + 4
        end
        error("invalid token")
    elseif c == "-" or (c >= "0" and c <= "9") then
        return parse_number(str, pos)
    else
        error("unexpected character " .. c)
    end
end

function M.decode(str)
    if str == nil then
        return nil
    end
    local value = parse_value(str, 1)
    return value
end

return M
