local EventBus = require 'event_bus'
local ResService = CS.LostRelic.ResService
local Object = CS.UnityEngine.Object
local GameObject = CS.UnityEngine.GameObject
local Vector3 = CS.UnityEngine.Vector3
local Quaternion = CS.UnityEngine.Quaternion
local ComponentFactory = CS.LostRelic.ComponentFactory
local EnemyAlertZoneType = CS.LostRelic.EnemyAlertZone

local M = {}
local enemies = {}
local player

local PATROL_SPEED = 1.8
local CHASE_SPEED = 4.0
local IDLE_MIN = 1.5
local IDLE_MAX = 4.0
local PATROL_RADIUS = 4.0
local CHASE_RADIUS_MULTIPLIER = 2.5
local ARRIVE_DISTANCE = 0.4
local ATTACK_DISTANCE = 1.2
local GRAVITY = 22
local PLAYER_ATTACK_RANGE = 2.5

-- A patrol leg is at most patrol_radius long (2m at 1.2 m/s for a guard turtle),
-- so anything past this means the agent is stuck rather than travelling: path
-- blocked, waypoint behind geometry, agent shoved off its path. The watchdog
-- sends it back to idle instead of letting it walk on the spot forever.
local PATROL_TIMEOUT = 6.0
-- Where in the swing the blow actually lands, as a fraction of the attack clip.
local ATTACK_HIT_FRACTION = 0.4

-- Death sink-out: a slain enemy holds its Die pose for DEATH_HOLD seconds, then
-- sinks DEATH_SINK_DEPTH world units straight into the ground over
-- DEATH_SINK_TIME seconds, after which the corpse is deactivated. The controller
-- and NavMeshAgent are already disabled the frame it dies (see apply_damage), and
-- both enemy prefabs have applyRootMotion=false, so writing transform.position
-- directly each frame is safe and nothing snaps the body back.
local DEATH_HOLD = 2.0
local DEATH_SINK_TIME = 1.0
local DEATH_SINK_DEPTH = 2.5

local function random_idle_time(enemy)
    local min_t = enemy ~= nil and enemy.idle_min or IDLE_MIN
    local max_t = enemy ~= nil and enemy.idle_max or IDLE_MAX
    return min_t + math.random() * (max_t - min_t)
end

local function play_anim(enemy, state_name)
    if enemy.animator ~= nil and enemy.current_anim ~= state_name then
        enemy.animator:CrossFade(state_name, 0.2, 0, 0)
        enemy.current_anim = state_name
    end
end

local function face_target(enemy, target)
    local direction = target - enemy.transform.position
    direction.y = 0
    if direction.magnitude > 0.1 then
        enemy.transform.rotation = Quaternion.LookRotation(direction.normalized)
    end
end

local function move_towards(enemy, target, speed, dt)
    local current = enemy.transform.position
    local direction = Vector3(target.x - current.x, 0, target.z - current.z)
    local distance = direction.magnitude
    if distance < ARRIVE_DISTANCE then
        return true
    end

    local motion = direction.normalized * speed
    motion.y = enemy.vertical_velocity
    enemy.controller:Move(motion * dt)

    if enemy.controller.isGrounded then
        enemy.vertical_velocity = -1
    else
        enemy.vertical_velocity = enemy.vertical_velocity - GRAVITY * dt
    end
    return false
end

local function stop_agent(enemy)
    if enemy.agent ~= nil and enemy.agent.isOnNavMesh then
        enemy.agent.isStopped = true
        enemy.agent:ResetPath()
    end
    enemy.nav_dest = nil
end

local function start_navigation(enemy, target, speed, stopping)
    if enemy.agent ~= nil and enemy.agent.isOnNavMesh then
        enemy.agent.isStopped = false
        enemy.agent.speed = speed
        enemy.agent.stoppingDistance = stopping
        -- Only re-issue the destination when it has actually moved. Calling
        -- SetDestination every frame keeps a path request permanently in
        -- flight, and while pathPending is true nav_arrived can never fire --
        -- that is what pinned the deep-ruins guards in WalkFWD for minutes.
        if enemy.nav_dest == nil or
            Vector3.Distance(enemy.nav_dest, target) > 0.15 then
            enemy.agent:SetDestination(target)
            enemy.nav_dest = target
        end
        return true
    end
    return false
end

local function nav_arrived(enemy)
    return enemy.agent ~= nil and enemy.agent.isOnNavMesh and
        not enemy.agent.pathPending and
        enemy.agent.remainingDistance <= enemy.agent.stoppingDistance + 0.2
end

function M.on_player_attack(damage)
    if player == nil then
        return
    end

    local player_pos = player.transform.position
    local best = nil
    local best_dist = PLAYER_ATTACK_RANGE
    for i = 1, #enemies do
        local enemy = enemies[i]
        if enemy ~= nil and not enemy.dead then
            local dist = Vector3.Distance(player_pos, enemy.transform.position)
            if dist < best_dist then
                best_dist = dist
                best = enemy
            end
        end
    end

    if best then
        M.apply_damage(best, damage)
        EventBus.publish("OnPlayerAttackHit", best.id)
    else
        EventBus.publish("OnPlayerAttackMiss")
    end
end

function M.apply_damage(enemy, raw_damage)
    if enemy == nil or enemy.dead then
        return
    end

    local amount = math.max(1, (raw_damage or 1) - enemy.defense)
    enemy.hp = enemy.hp - amount
    EventBus.publish("OnEnemyDamaged", enemy.id, amount)

        if enemy.hp <= 0 then
            enemy.hp = 0
            enemy.dead = true
            enemy.state = "dead"
            enemy.death_elapsed = 0
            EventBus.publish("OnEnemyHpHide", enemy.id)
            stop_agent(enemy)
            if enemy.controller then
                enemy.controller.enabled = false
            end
            if enemy.agent then
                enemy.agent.enabled = false
            end
            play_anim(enemy, "Die")
            EventBus.publish("OnEnemyDied", enemy.id)
        end
end

local function pick_waypoint(enemy)
    -- On a navmesh the waypoint must be a point the agent can finish a complete
    -- path to. A blind random offset lands behind walls often enough to matter
    -- (the deep ruins have one right west of the guards), and the agent answers
    -- an unreachable target with PathPartial plus a path request that never
    -- settles, so the patrol never completes. ComponentFactory rejects those
    -- candidates for us and returns the current position when boxed in, which
    -- reads as an instant arrival.
    if enemy.agent ~= nil and enemy.agent.isOnNavMesh then
        return ComponentFactory.SampleReachablePoint(
            enemy.transform.position,
            enemy.spawn,
            enemy.patrol_radius,
            8)
    end

    local angle = math.random() * math.pi * 2
    local radius = enemy.patrol_radius * (0.4 + 0.6 * math.random())
    return enemy.spawn + Vector3(
        math.cos(angle) * radius,
        0,
        math.sin(angle) * radius)
end

local function enter_patrol(enemy)
    enemy.state = "patrol"
    enemy.waypoint = pick_waypoint(enemy)
    enemy.patrol_elapsed = 0
    play_anim(enemy, "WalkFWD")
end

local function enter_idle(enemy)
    enemy.state = "idle"
    enemy.timer = random_idle_time(enemy)
    stop_agent(enemy)
    play_anim(enemy, "IdleNormal")
end

function M.init(root_go, player_go, enemy_configs)
    player = player_go
    enemies = {}

    for i = 1, #enemy_configs do
        local config = enemy_configs[i]
        local object_name = "遗迹守卫_" .. i
        local candidates = {}
        local seen = {}

        local function add_candidate(go)
            if go ~= nil and not seen[go:GetInstanceID()] then
                seen[go:GetInstanceID()] = true
                candidates[#candidates + 1] = go
            end
        end

        -- Reuse an instance already named for this slot (e.g. left over from a
        -- previous run); otherwise spawn a fresh one from the config prefab.
        -- We deliberately do NOT match by prefab/display name: several enemies
        -- can share one prefab, and partial-name matching would collapse them
        -- all onto a single scene object and skip spawning the rest.
        local exact_all = ComponentFactory.FindAllByExactName(object_name)
        if exact_all ~= nil then
            for j = 0, exact_all.Length - 1 do
                add_candidate(exact_all[j])
            end
        end

        if #candidates == 0 then
            local prefab = ResService.LoadPrefab(config.prefab)
            if prefab then
                local spawned = Object.Instantiate(prefab)
                spawned.name = object_name
                spawned.transform:SetParent(root_go.transform, true)
                local pos = config.position
                spawned.transform.position = Vector3(pos[1], pos[2], pos[3])
                spawned.transform.rotation = Quaternion.Euler(0, config.rotationY or 0, 0)
                add_candidate(spawned)
            end
        end

        for c = 1, #candidates do
            local go = candidates[c]
            local enemy_id = config.id or object_name

            CS.UnityEngine.Debug.Log(
                "[Enemy] found " .. go.name ..
                " id=" .. enemy_id ..
                " pos=" .. go.transform.position.x .. "," ..
                go.transform.position.y .. "," .. go.transform.position.z)

            local spawn_center = go.transform.position

            local zone = EnemyAlertZoneType.Attach(
                go,
                enemy_id,
                config.alertRadius or 7,
                config.patrolRadius or PATROL_RADIUS,
                config.chaseRadius or 17.5,
                config.patrolSpeed or PATROL_SPEED,
                config.chaseSpeed or CHASE_SPEED,
                config.idleMin or IDLE_MIN,
                config.idleMax or IDLE_MAX,
                config.attackDistance or ATTACK_DISTANCE,
                config.maxHp or 50,
                config.hp or 50,
                config.attack or 5,
                config.defense or 2,
                config.attackRange or 1.5,
                config.attackInterval or 1.2)

            -- The zone component is the tuning authority: everything numeric is
            -- read back off it so editing 遗迹守卫_N in the Inspector is what
            -- changes behaviour. EnemyAlertZone.Attach only seeds those fields
            -- from spawn_config.json when the component did not exist yet.
            local alert_radius = zone.radius
            if alert_radius <= 0 then
                alert_radius = 7
            end
            local chase_radius = zone.chaseRadius
            if chase_radius <= 0 then
                chase_radius = alert_radius * CHASE_RADIUS_MULTIPLIER
            end
            local max_hp = zone.maxHp
            if max_hp <= 0 then
                max_hp = 50
            end
            local start_hp = zone.hp
            if start_hp <= 0 then
                start_hp = max_hp
            end

            local enemy = {
                id = enemy_id,
                root = go,
                transform = go.transform,
                spawn = spawn_center,
                controller = ComponentFactory.AddCharacterController(
                    go,
                    1.2,
                    0.6,
                    Vector3(0, 0.6, 0)),
                agent = ComponentFactory.AddNavMeshAgent(
                    go,
                    0.6,
                    1.2,
                    PATROL_SPEED,
                    360),
                vertical_velocity = 0,
                animator = ComponentFactory.GetAnimatorInChildren(go),
                current_anim = "",
                state = "idle",
                timer = 0,
                waypoint = nil,
                alerted = false,
                alert_radius = alert_radius,
                chase_radius = chase_radius,
                patrol_radius = zone.patrolRadius,
                patrol_speed = zone.patrolSpeed,
                chase_speed = zone.chaseSpeed,
                idle_min = zone.idleMin,
                idle_max = zone.idleMax,
                attack_distance = zone.attackDistance,
                max_hp = max_hp,
                hp = start_hp,
                attack = zone.attack,
                defense = zone.defense,
                attack_range = zone.attackRange,
                attack_interval = zone.attackInterval,
                attack_cooldown = 0,
                dead = false,
                zone = zone
            }
            enemy.timer = random_idle_time(enemy)
            -- Swing timing comes from the clip itself, so one attack always
            -- reads as one swing no matter which enemy prefab this is. Both
            -- shipped controllers carry IdleBattle, the stance held between
            -- swings; IdleNormal is the fallback for anything that does not.
            enemy.attack_anim_len =
                ComponentFactory.GetClipLength(enemy.animator, "Attack01")
            if enemy.attack_anim_len <= 0 then
                enemy.attack_anim_len = 0.8
            end
            enemy.stance_anim = "IdleNormal"
            if ComponentFactory.GetClipLength(enemy.animator, "IdleBattle") > 0 then
                enemy.stance_anim = "IdleBattle"
            end
            enemies[#enemies + 1] = enemy

            -- The prefab ships with an already-enabled NavMeshAgent, which binds
            -- to the navmesh nearest the prefab origin the instant it spawns.
            -- Assigning transform.position afterwards does NOT move the agent,
            -- and with updatePosition=true it snaps the body back to that origin
            -- island on the next frame. Warp re-seats the agent's simulation at
            -- the real spawn so the enemy actually holds its encounter position.
            if enemy.agent ~= nil and enemy.agent.enabled then
                enemy.agent:Warp(spawn_center)
            end

            CS.UnityEngine.Debug.Log(
                "[Enemy] registered " .. enemy.id ..
                " hp=" .. enemy.hp .. "/" .. enemy.max_hp ..
                " atk=" .. enemy.attack ..
                " def=" .. enemy.defense ..
                " atkRange=" .. enemy.attack_range ..
                " atkDist=" .. enemy.attack_distance ..
                " alert=" .. enemy.alert_radius ..
                " chase=" .. enemy.chase_radius ..
                " (from Inspector)")
            stop_agent(enemy)
            play_anim(enemy, "IdleNormal")
        end
    end

    EventBus.subscribe("OnPlayerAttack", M.on_player_attack)
end

function M.update(dt)
    if player == nil or dt <= 0 then
        return
    end

    local player_position = player.transform.position
    for i = 1, #enemies do
        local enemy = enemies[i]
        if enemy.dead then
            if not enemy.removed then
                enemy.death_elapsed = (enemy.death_elapsed or 0) + dt
                if enemy.death_elapsed >= DEATH_HOLD then
                    -- Snapshot the resting spot the first sink frame so root-motion
                    -- drift (if any) can't move the anchor mid-sink.
                    if enemy.sink_base == nil then
                        enemy.sink_base = enemy.transform.position
                    end
                    local b = enemy.sink_base
                    local frac = (enemy.death_elapsed - DEATH_HOLD) / DEATH_SINK_TIME
                    if frac >= 1 then
                        enemy.transform.position =
                            Vector3(b.x, b.y - DEATH_SINK_DEPTH, b.z)
                        enemy.root:SetActive(false)
                        enemy.removed = true
                    else
                        enemy.transform.position =
                            Vector3(b.x, b.y - DEATH_SINK_DEPTH * frac, b.z)
                    end
                end
            end
        else
            local distance = Vector3.Distance(player_position, enemy.transform.position)
            local use_navmesh = enemy.agent ~= nil and enemy.agent.isOnNavMesh

            if distance <= enemy.alert_radius then
                if not enemy.alerted then
                    enemy.alerted = true
                    CS.UnityEngine.Debug.Log(
                        "[Enemy] alert " .. enemy.id ..
                        " dist=" .. distance ..
                        " alertRadius=" .. enemy.alert_radius)
                    EventBus.publish("OnEnemyAlert", enemy.id)
                    EventBus.publish(
                        "OnEnemyHpShow",
                        enemy.id,
                        enemy.hp,
                        enemy.max_hp,
                        enemy.root,
                        enemy.transform.position)
                end
                if enemy.state ~= "chase" then
                    enemy.state = "chase"
                    CS.UnityEngine.Debug.Log(
                        "[Enemy] chase " .. enemy.id ..
                        " dist=" .. distance)
                    play_anim(enemy, "RunFWD")
                end
            elseif enemy.alerted and distance > enemy.chase_radius then
                EventBus.publish("OnEnemyHpHide", enemy.id)
                enemy.alerted = false
                EventBus.publish("OnEnemyClear", enemy.id)
                enter_patrol(enemy)
            end

            if enemy.state == "chase" then
                if use_navmesh then
                    start_navigation(enemy, player_position, enemy.chase_speed, enemy.attack_distance)
                    if nav_arrived(enemy) then
                        face_target(enemy, player_position)
                    end
                else
                    face_target(enemy, player_position)
                    if distance > enemy.attack_distance then
                        move_towards(enemy, player_position, enemy.chase_speed, dt)
                    end
                end

                if distance <= enemy.attack_range and enemy.attack_cooldown <= 0 then
                    -- Start one swing. The blow itself is published partway in
                    -- (ATTACK_HIT_FRACTION) instead of on this frame, so what
                    -- the player sees and when the health bar drops line up.
                    enemy.attack_cooldown = enemy.attack_interval
                    enemy.swing_left = enemy.attack_anim_len
                    enemy.hit_left = enemy.attack_anim_len * ATTACK_HIT_FRACTION
                    if enemy.animator then
                        enemy.animator:Play("Attack01", 0, 0)
                        enemy.current_anim = "Attack01"
                    end
                elseif enemy.swing_left == nil then
                    -- Between swings the body has to match what it is doing.
                    -- Attack01 is a looping clip on TurtleShell, so leaving it
                    -- running is exactly what made the swings out-number the
                    -- hits (0.83s per swing against a 1.6s attack interval).
                    if distance <= enemy.attack_range then
                        play_anim(enemy, enemy.stance_anim)
                    else
                        play_anim(enemy, "RunFWD")
                    end
                end
            elseif enemy.state == "idle" then
                enemy.timer = enemy.timer - dt
                if enemy.timer <= 0 then
                    enter_patrol(enemy)
                end
            elseif enemy.state == "patrol" then
                enemy.patrol_elapsed = (enemy.patrol_elapsed or 0) + dt
                local arrived = false
                if use_navmesh then
                    start_navigation(enemy, enemy.waypoint, enemy.patrol_speed, ARRIVE_DISTANCE)
                    arrived = nav_arrived(enemy)
                else
                    face_target(enemy, enemy.waypoint)
                    arrived = move_towards(enemy, enemy.waypoint, enemy.patrol_speed, dt)
                end
                if arrived or enemy.patrol_elapsed >= PATROL_TIMEOUT then
                    enter_idle(enemy)
                end
            end

            -- Swing bookkeeping, outside the state branches so a swing still
            -- resolves if the enemy loses the player mid-attack.
            if enemy.hit_left ~= nil then
                enemy.hit_left = enemy.hit_left - dt
                if enemy.hit_left <= 0 then
                    enemy.hit_left = nil
                    EventBus.publish("OnPlayerDamaged", enemy.attack)
                end
            end
            if enemy.swing_left ~= nil then
                enemy.swing_left = enemy.swing_left - dt
                if enemy.swing_left <= 0 then
                    enemy.swing_left = nil
                    -- Chase re-picks its own clip next frame; the other states
                    -- only set theirs on entry, so re-assert them here.
                    if enemy.state == "patrol" then
                        play_anim(enemy, "WalkFWD")
                    elseif enemy.state == "idle" then
                        play_anim(enemy, "IdleNormal")
                    end
                end
            end

            enemy.attack_cooldown = math.max(0, enemy.attack_cooldown - dt)
            if enemy.alerted then
                EventBus.publish(
                    "OnEnemyHpUpdate",
                    enemy.id,
                    enemy.hp,
                    enemy.max_hp,
                    enemy.root,
                    enemy.transform.position)
            end
        end
    end
end

return M
