local EventBus = require 'event_bus'
local UIManager = CS.LostRelic.UIManager
local ComponentFactory = CS.LostRelic.ComponentFactory
local GameObject = CS.UnityEngine.GameObject
local Vector2 = CS.UnityEngine.Vector2
local Color = CS.UnityEngine.Color
local TextAnchor = CS.UnityEngine.TextAnchor

local M = {}
local font
local panel_sprite
local items_by_id = {}
local panel
local quest_name
local quest_desc
local progress_texts = {}
local total_text
local panel_open = false
local tab_inventory
local tab_attr
local tab_quest

local function log_missing(label, go)
    if go == nil then
        CS.UnityEngine.Debug.LogWarning("[UI] quest missing " .. label)
    end
end

function M.bind_scene_ui()
    local panel_go = ComponentFactory.FindIncludingInactive("QuestPanel")
    local name_go = ComponentFactory.FindIncludingInactive("QuestName")
    local desc_go = ComponentFactory.FindIncludingInactive("QuestDesc")
    local total_go = ComponentFactory.FindIncludingInactive("QuestTotal")
    local tab_inv_go = ComponentFactory.FindIncludingInactive("QuestTabInventory")
    local tab_attr_go = ComponentFactory.FindIncludingInactive("QuestTabAttr")
    local tab_quest_go = ComponentFactory.FindIncludingInactive("QuestTabActive")
    log_missing("QuestPanel", panel_go)
    log_missing("QuestName", name_go)
    log_missing("QuestDesc", desc_go)
    log_missing("QuestTotal", total_go)
    log_missing("QuestTabInventory", tab_inv_go)
    log_missing("QuestTabAttr", tab_attr_go)
    log_missing("QuestTabActive", tab_quest_go)
    if panel_go == nil or name_go == nil or desc_go == nil or
        total_go == nil or tab_inv_go == nil or
        tab_attr_go == nil or tab_quest_go == nil then
        return false
    end
    panel_go:SetActive(true)
    tab_inv_go:SetActive(true)
    tab_attr_go:SetActive(true)
    tab_quest_go:SetActive(true)

    panel = panel_go.transform
    quest_name = ComponentFactory.GetText(name_go.transform)
    quest_desc = ComponentFactory.GetText(desc_go.transform)
    total_text = ComponentFactory.GetText(total_go.transform)
    progress_texts = {}
    for i = 1, 4 do
        local go = ComponentFactory.FindIncludingInactive("QuestProgress" .. i)
        if go == nil then
            CS.UnityEngine.Debug.LogWarning("[UI] quest missing QuestProgress" .. i)
            return false
        end
        go:SetActive(true)
        progress_texts[i] = ComponentFactory.GetText(go.transform)
    end

    tab_inventory = tab_inv_go.transform
    tab_attr = tab_attr_go.transform
    tab_quest = tab_quest_go.transform

    local close_go = panel:Find("CloseQuestPanel")
    if close_go ~= nil then
        local close_comp = ComponentFactory.AddButton(close_go.gameObject)
        close_comp.onClick:AddListener(function()
            EventBus.publish("OnQuestPanelToggleRequest")
        end)
    end

    local inv_comp = ComponentFactory.AddButton(tab_inv_go)
    inv_comp.onClick:AddListener(function()
        EventBus.publish("OnPanelCloseRequest")
        EventBus.publish("OnPanelOpenRequest", "inventory")
    end)
    local attr_comp = ComponentFactory.AddButton(tab_attr_go)
    attr_comp.onClick:AddListener(function()
        EventBus.publish("OnPanelCloseRequest")
        EventBus.publish("OnPanelOpenRequest", "attr")
    end)
    ComponentFactory.AddButton(tab_quest_go)

    panel.gameObject:SetActive(false)
    tab_inventory.gameObject:SetActive(false)
    tab_attr.gameObject:SetActive(false)
    tab_quest.gameObject:SetActive(false)
    CS.UnityEngine.Debug.Log("[UI] quest scene ui bound")
    return true
end

function M.init(res_font, res_panel, config_items)
    font = res_font
    panel_sprite = res_panel
    items_by_id = config_items or {}
    if M.bind_scene_ui() then
        EventBus.subscribe("OnQuestSnapshot", M.update)
        return
    end
    CS.UnityEngine.Debug.LogWarning("[UI] quest fallback ui")
    local root = UIManager.Root

    panel = UIManager.CreatePanel(
        "QuestPanel", root, 860, 680,
        Color(0.08, 0.10, 0.13, 0.97))
    panel.anchoredPosition = Vector2(0, 0)
    UIManager.CreateImage(
        "QuestPanelBg", panel, panel_sprite,
        Color(1, 1, 1, 0.06), Vector2(0, 0), Vector2(840, 660))

    local header = UIManager.CreatePanel(
        "QuestHeader", panel, 820, 72,
        Color(0.14, 0.17, 0.22, 1))
    header.anchoredPosition = Vector2(0, 284)
    local accent = UIManager.CreatePanel(
        "QuestHeaderAccent", header, 6, 36,
        Color(0.92, 0.68, 0.30, 1))
    accent.anchoredPosition = Vector2(-400, 0)
    UIManager.CreateText(
        "QuestTitle", header, "任务", 28,
        Color(0.96, 0.84, 0.55, 1), font, TextAnchor.MiddleLeft,
        Vector2(-250, 0), Vector2(200, 40))

    quest_name = UIManager.CreateText(
        "QuestName", panel, "", 26,
        Color(0.95, 0.82, 0.5, 1), font, TextAnchor.MiddleLeft,
        Vector2(-180, 200), Vector2(520, 40))
    quest_desc = UIManager.CreateText(
        "QuestDesc", panel, "", 20,
        Color(0.88, 0.89, 0.92, 1), font, TextAnchor.UpperLeft,
        Vector2(-180, 120), Vector2(520, 70))

    local ys = { 20, -40, -100, -160 }
    for i = 1, 4 do
        progress_texts[i] = UIManager.CreateText(
            "QuestProgress" .. i, panel, "", 20,
            Color(0.92, 0.93, 0.95, 1), font, TextAnchor.MiddleLeft,
            Vector2(-180, ys[i]), Vector2(520, 36))
    end

    total_text = UIManager.CreateText(
        "QuestTotal", panel, "", 22,
        Color(0.92, 0.78, 0.42, 1), font, TextAnchor.MiddleLeft,
        Vector2(-180, -250), Vector2(400, 36))

    local close_button = UIManager.CreatePanel(
        "CloseQuestPanel", panel, 130, 48,
        Color(0.45, 0.22, 0.18, 1))
    close_button.anchoredPosition = Vector2(350, -300)
    local close_comp = ComponentFactory.AddButton(close_button.gameObject)
    close_comp.onClick:AddListener(function()
        EventBus.publish("OnQuestPanelToggleRequest")
    end)
    UIManager.CreateText(
        "CloseQuestText", close_button, "关闭", 22,
        Color(1, 1, 1, 1), font, TextAnchor.MiddleCenter,
        Vector2(0, 0), Vector2(130, 48))

    tab_inventory = UIManager.CreatePanel(
        "QuestTabInventory", root, 150, 60,
        Color(0.16, 0.22, 0.28, 1))
    tab_inventory.anchoredPosition = Vector2(-560, 150)
    local inv_tab_comp = ComponentFactory.AddButton(tab_inventory.gameObject)
    inv_tab_comp.onClick:AddListener(function()
        EventBus.publish("OnPanelCloseRequest")
        EventBus.publish("OnPanelOpenRequest", "inventory")
    end)
    UIManager.CreateText(
        "QuestTabInventoryText", tab_inventory, "背包面板", 20,
        Color(1, 1, 1, 1), font, TextAnchor.MiddleCenter,
        Vector2(0, 0), Vector2(150, 60))

    tab_attr = UIManager.CreatePanel(
        "QuestTabAttr", root, 150, 60,
        Color(0.16, 0.22, 0.28, 1))
    tab_attr.anchoredPosition = Vector2(-560, 60)
    local attr_tab_comp = ComponentFactory.AddButton(tab_attr.gameObject)
    attr_tab_comp.onClick:AddListener(function()
        EventBus.publish("OnPanelCloseRequest")
        EventBus.publish("OnPanelOpenRequest", "attr")
    end)
    UIManager.CreateText(
        "QuestTabAttrText", tab_attr, "属性面板", 20,
        Color(1, 1, 1, 1), font, TextAnchor.MiddleCenter,
        Vector2(0, 0), Vector2(150, 60))

    tab_quest = UIManager.CreatePanel(
        "QuestTabActive", root, 150, 60,
        Color(0.32, 0.44, 0.52, 1))
    tab_quest.anchoredPosition = Vector2(-560, -30)
    local quest_tab_comp = ComponentFactory.AddButton(tab_quest.gameObject)
    UIManager.CreateText(
        "QuestTabActiveText", tab_quest, "任务面板", 20,
        Color(1, 1, 1, 1), font, TextAnchor.MiddleCenter,
        Vector2(0, 0), Vector2(150, 60))

    panel.gameObject:SetActive(false)
    tab_inventory.gameObject:SetActive(false)
    tab_attr.gameObject:SetActive(false)
    tab_quest.gameObject:SetActive(false)
    EventBus.subscribe("OnQuestSnapshot", M.update)
end

function M.update(snapshot)
    if snapshot == nil then
        quest_name.text = "暂无任务"
        quest_desc.text = ""
        total_text.text = "进度 0/0"
        for i = 1, 4 do
            progress_texts[i].text = ""
        end
        return
    end

    quest_name.text = snapshot.name or ""
    quest_desc.text = snapshot.desc or ""
    total_text.text = "进度 " .. tostring(snapshot.done) .. "/" .. tostring(snapshot.total)
    for i = 1, 4 do
        local target = snapshot.targets and snapshot.targets[i]
        if target then
            local cfg = items_by_id[target.itemId]
            local name = cfg and cfg.name or target.itemId
            progress_texts[i].text = name .. "  " ..
                tostring(target.current) .. "/" .. tostring(target.count)
        else
            progress_texts[i].text = ""
        end
    end
end

function M.open()
    if panel then
        panel.gameObject:SetActive(true)
        panel_open = true
    end
    if tab_inventory then
        tab_inventory.gameObject:SetActive(true)
        tab_attr.gameObject:SetActive(true)
        tab_quest.gameObject:SetActive(true)
    end
end

function M.close()
    if panel then
        panel.gameObject:SetActive(false)
    end
    if tab_inventory then
        tab_inventory.gameObject:SetActive(false)
        tab_attr.gameObject:SetActive(false)
        tab_quest.gameObject:SetActive(false)
    end
    panel_open = false
end

function M.is_open()
    return panel_open
end

function M.shutdown()
    EventBus.unsubscribe("OnQuestSnapshot", M.update)
end

return M
