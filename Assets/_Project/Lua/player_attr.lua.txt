local EventBus = require 'event_bus'

local M = {}

-- The player's baseline numbers, and the only place they are written down.
-- spawn_config.json's player.stats block overrides these per key; anything it
-- leaves out falls back here. main.lua used to carry a second copy of the same
-- five numbers, which is how the JSON block and the code drift apart.
local DEFAULTS = {
    maxHp = 100,
    hp = 100,
    attack = 12,
    defense = 5,
    speed = 3.6
}

local attrs = {}
for key, value in pairs(DEFAULTS) do
    attrs[key] = value
end

function M.init(config)
    config = config or {}

    -- Per-key merge, the same rule the enemy channel uses: a key present in the
    -- JSON wins, a key omitted keeps the default. It has to be a nil test and
    -- not `or`, because 0 is a legitimate value (defense 0) and `or` would
    -- silently swap it back for the default.
    local from_config = {}
    for key, default in pairs(DEFAULTS) do
        local value = config[key]
        if value ~= nil then
            attrs[key] = value
            from_config[#from_config + 1] = key
        else
            attrs[key] = default
        end
    end

    -- Raising maxHp without restating hp means "starts at full", so 100 -> 150
    -- does not put the player at 100/150. Same rule as the enemies.
    if config.hp == nil then
        attrs.hp = attrs.maxHp
    end

    -- A misspelt key would otherwise do nothing whatsoever, with no clue why the
    -- edit had no effect, so name it rather than ignore it.
    local unknown = {}
    for key in pairs(config) do
        if DEFAULTS[key] == nil then
            unknown[#unknown + 1] = tostring(key)
        end
    end

    table.sort(from_config)
    table.sort(unknown)

    local source
    if #from_config == 0 then
        source = " (all defaults)"
    else
        source = " (spawn_config.json: " .. table.concat(from_config, ",") .. ")"
    end
    if #unknown > 0 then
        source = source .. "  !! unrecognised key(s) ignored: " ..
            table.concat(unknown, ",")
    end

    -- defense is deliberately reported even though nothing consumes it yet:
    -- take_damage subtracts the raw amount, so the number is panel decoration.
    CS.UnityEngine.Debug.Log(
        "[Player] stats hp=" .. attrs.hp .. "/" .. attrs.maxHp ..
        " atk=" .. attrs.attack ..
        " def=" .. attrs.defense .. "(unused)" ..
        " speed=" .. attrs.speed ..
        source)

    M.publish()
end

function M.get()
    return {
        maxHp = attrs.maxHp,
        hp = attrs.hp,
        attack = attrs.attack,
        defense = attrs.defense,
        speed = attrs.speed
    }
end

function M.get_speed()
    return attrs.speed
end

function M.take_damage(amount)
    if amount == nil or amount <= 0 then
        return
    end
    attrs.hp = math.max(0, attrs.hp - amount)
    M.publish()
end

function M.heal(amount)
    if amount == nil or amount <= 0 then
        return
    end
    attrs.hp = math.min(attrs.maxHp, attrs.hp + amount)
    M.publish()
end

function M.publish()
    EventBus.publish("OnPlayerAttributeChanged", M.get())
end

return M
